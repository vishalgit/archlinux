(defun my/leptosfmt-buffer ()
  "Format current buffer with leptosfmt, then rustfmt"
  (interactive)
  (when (and (eq major-mode 'rustic-mode)
             (save-excursion
               (goto-char (point-min))
               (re-search-forward "view!" nil t)))
    (let ((file (buffer-file-name)))
      (when file
        (save-buffer)
        (call-process "leptosfmt" nil nil nil file)
        (revert-buffer t t t)))))
(add-hook 'rustic-mode-hook
          (lambda ()
            (add-hook 'before-save-hook #'my/leptosfmt-buffer nil t)))
#+end_src

;; C# auto find dll
(defun my/dap-netcore-auto-program ()
  "Auto detect the dotnet dll for debugging"
  (let ((dll (car (file-expand-wildcards
                   (expand-file-name "bin/Debug/*/*.dll" (projectile-project-root))))))
    (when dll
      (setq-local dap-netcore-program dll ))))
;; Rust auto-find binary
(defun my/dap-rust-auto-program ()
  "Auto detect the rust binary for debugging"
  (let ((bin (expand-file-name
              (concat "target/debug/" (file-name-nondirectory (projectile-project-root)))
              (projectile-project-root))))
    (when (file-exists-p bin)
      (setq-local dap-lldb-program bin))))

#+begin_src emacs-lisp :tangle "~/.config/doom/config.el"
(defvar my/blog-dir "~/org/vishalgit.github.io/")

(defun my/blog-publish ()
  "Export the whole blog to HTML locally."
  (interactive)
  (let ((default-directory my/blog-dir))
    (load-file (expand-file-name "publish.el" my/blog-dir))))

(defun my/blog-new-post (title)
  "Create a new dated org post."
  (interactive "sPost title: ")
  (let* ((slug (downcase (replace-regexp-in-string "[^a-z0-9]+" "-" title)))
         (file (format "%scontent/posts/%s-%s.org"
                       my/blog-dir (format-time-string "%Y-%m-%d") slug)))
    (find-file file)
    (insert (format "#+TITLE: %s\n#+DATE: %s\n#+OPTIONS: toc:nil num:nil\n\n" title (format-time-string "%Y-%m-%d")))))
